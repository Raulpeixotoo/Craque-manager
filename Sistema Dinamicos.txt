class Narrador {
  constructor(time, adversario, local) {
    this.time = time;
    this.adversario = adversario;
    this.local = local;
    this.narrador = this.escolherNarrador();
    this.frasesUsadas = [];
    this.ultimaFrase = '';
    this.emocao = 0.5;
    this.eventos = [];
    this.contexto = this.gerarContexto();
  }
  
  escolherNarrador() {
    const nomes = Object.keys(NARRADORES).filter(n => n !== 'narrador_aleatorio');
    const nome = pick(nomes);
    return NARRADORES[nome];
  }
  
  gerarContexto() {
    const climas = Object.keys(CONTEXTOS.clima);
    const tiposTorcida = Object.keys(CONTEXTOS.torcida);
    return {
      clima: pick(climas),
      torcida: pick(tiposTorcida),
      minuto: 0,
      fase: 'inicio',
      momento: 'normal', // 'normal', 'decisivo', 'classico'
      placar: { casa: 0, fora: 0 }
    };
  }
  
  narrar(evento) {
    this.eventos.push(evento);
    this.atualizarContexto(evento);
    
    let frase = this.gerarFrase(evento);
    frase = this.aplicarEmocao(frase);
    frase = this.aplicarContexto(frase);
    frase = this.adicionarBordao(frase);
    
    this.ultimaFrase = frase;
    return this.formatarFrase(frase, evento);
  }
  
  gerarFrase(evento) {
    const { tipo, dados } = evento;
    let frases = [];
    let substitutos = {};
    
    switch(tipo) {
      case 'gol':
        frases = this.selecionarFrases('gol', dados);
        substitutos = {
          time: this.time.nome,
          jogador: dados.jogador,
          adversario: this.adversario.nome
        };
        break;
        
      case 'finalizacao':
        frases = this.selecionarFrases('finalizacao', dados);
        substitutos = {
          jogador: dados.jogador,
          time: this.time.nome,
          goleiro: dados.goleiro || 'goleiro',
          adversario: this.adversario.nome
        };
        break;
        
      case 'cartao':
        frases = this.selecionarFrases('cartao', dados);
        substitutos = {
          jogador: dados.jogador,
          time: this.time.nome,
          cor: dados.cor || 'amarelo'
        };
        break;
        
      case 'substituicao':
        frases = FRASES.substituicao;
        substitutos = {
          entra: dados.entra,
          sai: dados.sai,
          time: this.time.nome
        };
        break;
        
      case 'lesao':
        frases = FRASES.lesao;
        substitutos = {
          jogador: dados.jogador,
          time: this.time.nome
        };
        break;
        
      case 'perigo':
        frases = FRASES.perigo;
        substitutos = {
          time: this.time.nome,
          adversario: this.adversario.nome,
          jogador: dados.jogador || 'jogador'
        };
        break;
        
      case 'impedimento':
        frases = FRASES.impedimento;
        substitutos = {
          jogador: dados.jogador,
          time: this.time.nome
        };
        break;
        
      case 'escanteio':
        frases = FRASES.escanteio;
        substitutos = {
          time: this.time.nome,
          jogador: dados.jogador || 'jogador'
        };
        break;
        
      case 'falta':
        frases = FRASES.falta;
        substitutos = {
          jogador: dados.jogador,
          time: this.time.nome
        };
        break;
        
      case 'inicio':
        frases = this.contexto.classico ? FRASES.classico : FRASES.inicio;
        substitutos = {
          time: this.time.nome,
          adversario: this.adversario.nome
        };
        break;
        
      case 'intervalo':
        frases = FRASES.intervalo;
        substitutos = {
          time: this.time.nome,
          adversario: this.adversario.nome,
          gols_casa: this.contexto.placar.casa,
          gols_fora: this.contexto.placar.fora
        };
        break;
        
      case 'fim':
        frases = FRASES.fim;
        substitutos = {
          time: this.time.nome,
          adversario: this.adversario.nome,
          gols_casa: this.contexto.placar.casa,
          gols_fora: this.contexto.placar.fora
        };
        break;
        
      case 'posse':
        frases = FRASES.posse;
        substitutos = {
          time: dados.time || this.time.nome
        };
        break;
        
      default:
        frases = ['Lance importante no jogo!'];
        substitutos = {};
    }
    
    if (!frases || frases.length === 0) {
      frases = ['Momento intenso no jogo!'];
    }
    
    // Evitar repetição de frases
    let frase = this.escolherFraseUnica(frases);
    
    // Substituir placeholders
    for (let [key, value] of Object.entries(substitutos)) {
      frase = frase.replace(`{${key}}`, value);
    }
    
    return frase;
  }
  
  selecionarFrases(categoria, dados) {
    const subcategorias = FRASES[categoria];
    if (!subcategorias) return ['Evento importante!'];
    
    if (dados.contexto === 'decisivo' && subcategorias.decisivos) {
      return subcategorias.decisivos;
    }
    if (dados.contexto === 'contra' && subcategorias.contra) {
      return subcategorias.contra;
    }
    if (dados.emocao && dados.emocao > 0.7 && subcategorias.emocionados) {
      return subcategorias.emocionados;
    }
    
    // Subcategoria específica
    if (dados.subcategoria && subcategorias[dados.subcategoria]) {
      return subcategorias[dados.subcategoria];
    }
    
    // Fallback para a subcategoria principal
    const chaves = Object.keys(subcategorias);
    const principal = chaves.find(k => ['simples', 'normal'].includes(k));
    return subcategorias[principal] || subcategorias[chaves[0]];
  }
  
  escolherFraseUnica(frases) {
    // Evita repetir a mesma frase nos últimos 5 eventos
    const disponiveis = frases.filter(f => !this.frasesUsadas.includes(f));
    let frase;
    
    if (disponiveis.length > 0) {
      frase = pick(disponiveis);
    } else {
      frase = pick(frases);
    }
    
    this.frasesUsadas.push(frase);
    if (this.frasesUsadas.length > 10) {
      this.frasesUsadas.shift();
    }
    
    return frase;
  }
  
  aplicarEmocao(frase) {
    // Aumenta a emoção baseado no contexto do jogo
    const emocao = this.calcularEmocao();
    
    if (emocao > 0.8) {
      // Frase com mais emoção
      if (Math.random() < 0.4) {
        frase = frase.toUpperCase() + '!';
      }
    }
    
    return frase;
  }
  
  calcularEmocao() {
    const placar = this.contexto.placar;
    const diferenca = Math.abs(placar.casa - placar.fora);
    const total = placar.casa + placar.fora;
    
    // Jogos apertados são mais emocionantes
    if (total > 0 && diferenca <= 1) {
      this.emocao = Math.min(1, this.emocao + 0.1);
    }
    
    // Jogos com muitos gols são emocionantes
    if (total >= 3) {
      this.emocao = Math.min(1, this.emocao + 0.1);
    }
    
    // Final do jogo aumenta emoção
    if (this.contexto.minuto > 80) {
      this.emocao = Math.min(1, this.emocao + 0.2);
    }
    
    // Clássico é sempre emocionante
    if (this.contexto.classico) {
      this.emocao = Math.min(1, this.emocao + 0.3);
    }
    
    // Jogo decisivo
    if (this.contexto.decisivo) {
      this.emocao = Math.min(1, this.emocao + 0.2);
    }
    
    return this.emocao;
  }
  
  aplicarContexto(frase) {
    const ctx = this.contexto;
    
    // Adicionar contexto em momentos específicos
    if (ctx.minuto === 0 && ctx.clima) {
      frase = CONTEXTOS.clima[ctx.clima] + ' ' + frase;
    }
    
    if (ctx.minuto === 0 && ctx.torcida) {
      const torcida = CONTEXTOS.torcida[ctx.torcida];
      if (torcida) frase = frase + ' ' + torcida;
    }
    
    return frase;
  }
  
  adicionarBordao(frase) {
    // Adiciona bordões característicos do narrador
    if (this.narrador && this.narrador.bordoes && Math.random() < 0.15) {
      const bordao = pick(this.narrador.bordoes);
      frase = frase + ' ' + bordao;
    }
    return frase;
  }
  
  formatarFrase(frase, evento) {
    // Adicionar timestamp
    const minuto = evento.minuto || this.contexto.minuto;
    const prefixo = `[${minuto}'] `;
    
    // Adicionar nome do narrador em eventos importantes
    if (['gol', 'vermelho', 'fim'].includes(evento.tipo)) {
      return `${prefixo}🎙️ ${this.narrador.nome}: ${frase}`;
    }
    
    return `${prefixo}${frase}`;
  }
  
  atualizarContexto(evento) {
    if (evento.minuto !== undefined) {
      this.contexto.minuto = evento.minuto;
    }
    
    if (evento.tipo === 'gol') {
      if (evento.dados.time === this.time.id) {
        this.contexto.placar.casa++;
      } else {
        this.contexto.placar.fora++;
      }
    }
    
    if (evento.tipo === 'intervalo') {
      this.contexto.fase = 'intervalo';
    }
    
    if (evento.tipo === 'fim') {
      this.contexto.fase = 'fim';
    }
    
    // Detectar clássico
    if (this.time.rivalId === this.adversario.id || this.adversario.rivalId === this.time.id) {
      this.contexto.classico = true;
    }
    
    // Detectar jogo decisivo (fase de mata-mata)
    if (evento.dados && evento.dados.decisivo) {
      this.contexto.decisivo = true;
    }
  }
}

// Modificações no motor.js para incluir narração

function criarPartida(state, jogo) {
  const timeCasa = state.times[jogo.casa];
  const timeFora = state.times[jogo.fora];
  
  // Criar narrador
  const narrador = new Narrador(timeCasa, timeFora, 'casa');
  
  const partida = {
    g: jogo,
    H: forcaTime(state, jogo.casa, true),
    A: forcaTime(state, jogo.fora, false),
    gc: 0,
    gf: 0,
    min: 0,
    posse: 50,
    fc: 0,
    ff: 0,
    eventos: [],
    fim: false,
    narrador: narrador,
    narracao: [],
    narracaoAtual: ''
  };
  
  // Narração de início
  const inicioEvento = {
    tipo: 'inicio',
    minuto: 0,
    dados: {
      time: timeCasa.id,
      adversario: timeFora.id
    }
  };
  const fraseInicio = narrador.narrar(inicioEvento);
  partida.narracao.push(fraseInicio);
  partida.narracaoAtual = fraseInicio;
  
  return partida;
}

// Função para narrar eventos durante o jogo
function narrarEvento(partida, evento) {
  const narrador = partida.narrador;
  const frase = narrador.narrar(evento);
  partida.narracao.push(frase);
  partida.narracaoAtual = frase;
  partida.eventos.push({
    ...evento,
    narracao: frase
  });
  
  // Adicionar ao log do jogo
  if (evento.tipo === 'gol') {
    noticia(state, `⚽ ${evento.dados.jogador} marcou! ${frase}`);
  }
}

📝 Gerador de Resumo da Partida

function gerarResumoPartida(partida) {
  const narracao = partida.narracao;
  const gols = partida.eventos.filter(e => e.tipo === 'gol');
  const cartoes = partida.eventos.filter(e => e.tipo === 'cartao');
  
  let resumo = `
    🏆 ${partida.g.casaNome} ${partida.gc} x ${partida.gf} ${partida.g.foraNome}
    
    ⚽ Gols:
  `;
  
  gols.forEach(g => {
    resumo += `   ${g.dados.minuto}' - ${g.dados.jogador} (${g.dados.time === partida.g.casa ? partida.g.casaNome : partida.g.foraNome})\n`;
  });
  
  if (cartoes.length) {
    resumo += `\n🟨 Cartões:\n`;
    cartoes.forEach(c => {
      resumo += `   ${c.dados.minuto}' - ${c.dados.jogador} (${c.dados.cor})\n`;
    });
  }
  
  resumo += `\n📊 Posse: ${partida.posse}% x ${100 - partida.posse}%`;
  resumo += `\n🎯 Finalizações: ${partida.fc} x ${partida.ff}`;
  
  // Momentos marcantes da narração
  resumo += `\n\n🎙️ Momentos marcantes:\n`;
  const momentos = narracao.filter(f => 
    f.includes('GOOOOOOO') || 
    f.includes('GOL!') || 
    f.includes('INACREDITÁVEL') ||
    f.includes('classico') ||
    f.includes('decisivo')
  ).slice(-5);
  
  momentos.forEach(m => {
    resumo += `   ${m}\n`;
  });
  
  return resumo;
}
